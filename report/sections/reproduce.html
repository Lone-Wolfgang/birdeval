Everything on this page is generated from the repository. From a checkout:

uv pip install -e ".[models,perch,birdset,report,dev]"

# clips mode (BirdSet's protocol) and soundscape mode, one site at a time
for s in HSN NBP NES PER POW SNE SSW UHH; do
  birdeval run -m birdnet_2.4 birdnet_3.0 perch_v2 -e birdset_$s --run-name full_$s
  birdeval run -m birdnet_2.4 birdnet_3.0 perch_v2 -e birdset_${s}_scape --run-name full_${s}_scape
done

python -I scripts/build_report.py results/full_*      # report tables -> results/report/
python -I scripts/hourly.py                            # by hour of day, from the cached predictions
python -I scripts/fig_spectrogram.py                   # the figure in 2.1
python -I scripts/render_report.py                     # this page -> docs/index.html

Predictions are cached per file, so an interrupted run resumes where it stopped, and rebuilding the tables or the page never re-runs a model.
